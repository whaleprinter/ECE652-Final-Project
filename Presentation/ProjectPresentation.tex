\begin{frame}
  \frametitle{Coherence Protocol}
  
  \begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!black]
    \begin{center}
      \textbf{MSI Snooping Protocol} implemented with a shared bus 
    \end{center}
  \end{tcolorbox}

  \begin{itemize} 
    \item Independent L1I caches preloaded with instructions
    \item Miss in L1Ds \& Access L2: 6 cycles
    \item Get data from other core: 4 cycles
    \item Hit in local L1D: 1 cycle 
  \end{itemize}

  \begin{tcolorbox}[colback=PMS 166!15!white,colframe=PMS 166!100!white]
    \begin{center}
      \textbf{Consistency:} RISC-V follows release consistency, but we implement sequential consistency
    \end{center}
  \end{tcolorbox}

\end{frame}

\begin{frame} 
  \frametitle{Testing}
  Microbenchmarks:
  \begin{itemize}
    \item Independent, simultaneous core operation
    \item Basic sharing: \textsc{core 0} writes $\Rightarrow$ \textsc{core 1} reads
    \item Cores obey coherence protocol on repeated writes/reads
    \item Fast-paced false-sharing 
    \item Cache block evicted and fetched from L2 
    \item \textsc{core 0} populates array, \textsc{core 1} calculates sum
  \end{itemize}
